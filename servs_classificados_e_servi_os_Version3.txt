    <a class="nav-link active" data-target="homeView"><i class="fa-solid fa-house"></i><span>Início</span></a>
    <a class="nav-link" data-target="favoritesView"><i class="fa-solid fa-heart"></i><span>Favoritos</span></a>
    <a class="nav-link announce-btn" data-target="announceView"><i class="fa-solid fa-plus"></i></a>
    <a class="nav-link" data-target="chatView"><i class="fa-solid fa-comments"></i><span>Chat</span></a>
    <a class="nav-link" data-target="profileView"><i class="fa-solid fa-user"></i><span>Perfil</span></a>
  </nav>

  <script>
    const STORAGE_KEYS = {
      user: 'servs_user',
      favorites: 'servs_favorites',
      location: 'servs_location_filter',
      listings: 'servs_listings',
      theme: 'servs_theme'
    };

    const state = {
      usuarioAtual: null,
      anuncios: [],
      favoritos: [],
      fotosSelecionadas: [],
      categoriaAtual: 'Todas',
      localizacao: {
        estado: 'TODOS',
        cidade: 'TODAS',
        bairroArea: '',
        raioKm: 0
      },
      theme: localStorage.getItem(STORAGE_KEYS.theme) || 'light'
    };

    const ESTADOS_BRASIL = [
      { sigla: "AC", nome: "Acre" }, { sigla: "AL", nome: "Alagoas" }, { sigla: "AP", nome: "Amapá" },
      { sigla: "AM", nome: "Amazonas" }, { sigla: "BA", nome: "Bahia" }, { sigla: "CE", nome: "Ceará" },
      { sigla: "DF", nome: "Distrito Federal" }, { sigla: "ES", nome: "Espírito Santo" }, { sigla: "GO", nome: "Goiás" },
      { sigla: "MA", nome: "Maranhão" }, { sigla: "MT", nome: "Mato Grosso" }, { sigla: "MS", nome: "Mato Grosso do Sul" },
      { sigla: "MG", nome: "Minas Gerais" }, { sigla: "PA", nome: "Pará" }, { sigla: "PB", nome: "Paraíba" },
      { sigla: "PR", nome: "Paraná" }, { sigla: "PE", nome: "Pernambuco" }, { sigla: "PI", nome: "Piauí" },
      { sigla: "RJ", nome: "Rio de Janeiro" }, { sigla: "RN", nome: "Rio Grande do Norte" }, { sigla: "RS", nome: "Rio Grande do Sul" },
      { sigla: "RO", nome: "Rondônia" }, { sigla: "RR", nome: "Roraima" }, { sigla: "SC", nome: "Santa Catarina" },
      { sigla: "SP", nome: "São Paulo" }, { sigla: "SE", nome: "Sergipe" }, { sigla: "TO", nome: "Tocantins" }
    ];

    function safeParseJSON(key, fallback) {
      try {
        const raw = localStorage.getItem(key);
        return raw ? JSON.parse(raw) : fallback;
      } catch {
        return fallback;
      }
    }

    function normalizarTexto(value = '') {
      return String(value)
        .normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '')
        .trim()
        .toLowerCase();
    }

    function gerarId() {
      if (window.crypto && crypto.randomUUID) return crypto.randomUUID();
      return 'id-' + Date.now() + '-' + Math.random().toString(16).slice(2);
    }

    function formatarPreco(valor) {
      return Number(valor || 0).toLocaleString('pt-BR', {
        style: 'currency',
        currency: 'BRL'
      });
    }

    function aplicarTema(theme) {
      const body = document.body;
      const toggle = document.getElementById('themeToggle');

      state.theme = theme;
      body.setAttribute('data-theme', theme);

      if (toggle) {
        toggle.innerHTML = theme === 'dark'
          ? '<i class="fa-solid fa-sun"></i>'
          : '<i class="fa-solid fa-moon"></i>';
      }

      localStorage.setItem(STORAGE_KEYS.theme, theme);
    }

    function inicializarFiltroLocalizacao() {
      const selectEstado = document.getElementById('selectEstado');
      if (!selectEstado) return;

      selectEstado.innerHTML = '<option value="TODOS">Brasil (Todo o país)</option>';
      ESTADOS_BRASIL.forEach(uf => {
        selectEstado.innerHTML += `<option value="${uf.sigla}">${uf.nome} (${uf.sigla})</option>`;
      });

      selectEstado.value = state.localizacao.estado || 'TODOS';
      atualizarHeaderLocalizacaoText();
    }

    async function carregarCidades(uf) {
      const selectCidade = document.getElementById('selectCidade');
      if (!selectCidade) return;

      selectCidade.innerHTML = '<option value="TODAS">Todas as cidades</option>';

      if (uf === 'TODOS' || !uf) {
        selectCidade.disabled = true;
        return;
      }

      selectCidade.disabled = false;

      try {
        const res = await fetch(`https://servicodados.ibge.gov.br/api/v1/localidades/estados/${uf}/municipios`);
        const cidades = await res.json();

        cidades.forEach(cidade => {
          selectCidade.innerHTML += `<option value="${cidade.nome}">${cidade.nome}</option>`;
        });

        if (state.localizacao.cidade && state.localizacao.cidade !== 'TODAS') {
          selectCidade.value = state.localizacao.cidade;
        }
      } catch (err) {
        console.warn('Erro ao carregar cidades via IBGE:', err);
      }
    }

    function abrirModalLocalizacao() {
      const modal = document.getElementById('locationModalOverlay');
      if (!modal) return;

      const selectEstado = document.getElementById('selectEstado');
      const selectCidade = document.getElementById('selectCidade');
      const inputBairro = document.getElementById('inputBairroArea');
      const inputRaio = document.getElementById('inputRaio');

      if (selectEstado) selectEstado.value = state.localizacao.estado || 'TODOS';
      if (inputBairro) inputBairro.value = state.localizacao.bairroArea || '';
      if (inputRaio) inputRaio.value = state.localizacao.raioKm || 0;

      atualizarLabelRaio(state.localizacao.raioKm || 0);

      carregarCidades(state.localizacao.estado).then(() => {
        if (state.localizacao.cidade && selectCidade) {
          selectCidade.value = state.localizacao.cidade;
        }
      });

      modal.classList.add('active');
    }

    function fecharModalLocalizacao() {
      document.getElementById('locationModalOverlay')?.classList.remove('active');
    }

    function atualizarLabelRaio(valor) {
      const label = document.getElementById('labelRaio');
      if (!label) return;

      const val = Number(valor || 0);
      if (val === 0) {
        label.textContent = 'Apenas na área definida';
      } else {
        label.textContent = `+${val} km ao redor`;
      }
    }

    function salvarFiltroLocalizacao(e) {
      e.preventDefault();

      state.localizacao = {
        estado: document.getElementById('selectEstado')?.value || 'TODOS',
        cidade: document.getElementById('selectCidade')?.value || 'TODAS',
        bairroArea: document.getElementById('inputBairroArea')?.value.trim() || '',
        raioKm: Number(document.getElementById('inputRaio')?.value || 0)
      };

      localStorage.setItem(STORAGE_KEYS.location, JSON.stringify(state.localizacao));
      atualizarHeaderLocalizacaoText();
      fecharModalLocalizacao();
      renderizarAnuncios();
      renderizarFavoritos();
    }

    function atualizarHeaderLocalizacaoText() {
      const el = document.getElementById('currentLocationText');
      if (!el) return;

      let texto = 'Brasil (Todo o país)';

      if (state.localizacao.estado !== 'TODOS') {
        texto = state.localizacao.estado;

        if (state.localizacao.cidade !== 'TODAS') {
          texto = `${state.localizacao.cidade}, ${state.localizacao.estado}`;
        }

        if (state.localizacao.bairroArea) {
          texto += ` (${state.localizacao.bairroArea})`;
        }

        if (state.localizacao.raioKm > 0) {
          texto += ` +${state.localizacao.raioKm}km`;
        }
      }

      el.textContent = texto;
    }

    function anuncioBaseExemplo() {
      return {
        id: gerarId(),
        tipo: 'Serviço',
        titulo: 'Instalação de ar-condicionado',
        categoria: 'Eletricista',
        preco: 280,
        descricao: 'Instalação e manutenção de ar-condicionado residencial com garantia.',
        contato: '(68) 99999-8888',
        cidade: 'Rio Branco',
        estado: 'AC',
        bairroArea: 'Centro',
        fotos: [
          'https://images.unsplash.com/photo-1581578731548-c64695cc6952?auto=format&fit=crop&w=900&q=80'
        ],
        autor: 'demo@servs.com',
        criadoEm: new Date().toISOString()
      };
    }

    function carregarDadosIniciais() {
      const listaSalva = safeParseJSON(STORAGE_KEYS.listings, []);

      if (Array.isArray(listaSalva) && listaSalva.length) {
        state.anuncios = listaSalva;
        return;
      }

      state.anuncios = [anuncioBaseExemplo()];
      localStorage.setItem(STORAGE_KEYS.listings, JSON.stringify(state.anuncios));
    }

    function alternarFavorito(anuncioId) {
      const existe = state.favoritos.includes(anuncioId);

      if (existe) {
        state.favoritos = state.favoritos.filter(id => id !== anuncioId);
      } else {
        state.favoritos.push(anuncioId);
      }

      localStorage.setItem(STORAGE_KEYS.favorites, JSON.stringify(state.favoritos));
      renderizarAnuncios();
      renderizarFavoritos();
    }

    function obterAnunciosFiltrados() {
      const termo = normalizarTexto(document.getElementById('mainSearchInput')?.value || '');

      return state.anuncios.filter((anuncio) => {
        const textoGeral = normalizarTexto(
          `${anuncio.titulo || ''} ${anuncio.categoria || ''} ${anuncio.descricao || ''} ${anuncio.cidade || ''} ${anuncio.estado || ''} ${anuncio.bairroArea || ''}`
        );

        const atendeBusca = !termo || textoGeral.includes(termo);

        const categoriaValida =
          state.categoriaAtual === 'Todas' || (anuncio.categoria || 'Outros') === state.categoriaAtual;

        if (!categoriaValida) return false;

        if (state.localizacao.estado !== 'TODOS') {
          const estadoAnuncio = String(anuncio.estado || '').toUpperCase();

          if (estadoAnuncio && estadoAnuncio !== state.localizacao.estado && state.localizacao.raioKm === 0) {
            return false;
          }

          if (state.localizacao.cidade !== 'TODAS') {
            const cidadeAnuncio = normalizarTexto(anuncio.cidade || '');
            const cidadeFiltro = normalizarTexto(state.localizacao.cidade);

            if (state.localizacao.raioKm === 0 && !cidadeAnuncio.includes(cidadeFiltro)) {
              return false;
            }
          }

          if (state.localizacao.bairroArea) {
            const areaFiltro = normalizarTexto(state.localizacao.bairroArea);
            const areaAnuncio = normalizarTexto(`${anuncio.bairroArea || ''} ${anuncio.cidade || ''}`);

            if (state.localizacao.raioKm === 0 && !areaAnuncio.includes(areaFiltro)) {
              return false;
            }
          }
        }

        return atendeBusca;
      });
    }

    function renderizarAnuncios() {
      const publicListings = document.getElementById('publicListings');
      if (!publicListings) return;

      const filtrados = obterAnunciosFiltrados();

      publicListings.innerHTML = '';

      if (!filtrados.length) {
        publicListings.innerHTML = `
          <div class="empty-listings" style="grid-column:1/-1;">
            <i class="fa-solid fa-store" style="font-size:25px;display:block;margin-bottom:8px;"></i>
            Nenhum anúncio encontrado nesta região, categoria ou busca.
            <br>
            <small style="color:var(--text-soft);">Experimente outra categoria ou amplie o raio de busca.</small>
          </div>
        `;
        return;
      }

      filtrados.forEach((anuncio) => {
        const card = document.createElement('article');
        card.className = 'listing-card';

        const fotoPrincipal = Array.isArray(anuncio.fotos) && anuncio.fotos.length
          ? anuncio.fotos[0]
          : 'https://images.unsplash.com/photo-1581578731548-c64695cc6952?auto=format&fit=crop&w=900&q=80';

        const favoritado = state.favoritos.includes(anuncio.id);

        card.innerHTML = `
          <div class="listing-photo-container">
            <button class="fav-btn ${favoritado ? 'active' : ''}" type="button" aria-label="Favoritar anúncio">
              <i class="fa-${favoritado ? 'solid' : 'regular'} fa-heart"></i>
            </button>

            <img class="listing-photo" src="${fotoPrincipal}" alt="${anuncio.titulo || 'Anúncio'}">
          </div>

          <div class="listing-body">
            <div class="listing-price">${formatarPreco(anuncio.preco)}</div>
            <div class="listing-title">${anuncio.titulo || 'Anúncio sem título'}</div>
            <div class="listing-meta">${anuncio.cidade || 'Local'} • ${anuncio.estado || 'BR'}</div>
            <div class="listing-desc">${anuncio.descricao || 'Sem descrição disponível.'}</div>
          </div>
        `;

        card.querySelector('.fav-btn')?.addEventListener('click', () => {
          alternarFavorito(anuncio.id);
        });

        publicListings.appendChild(card);
      });
    }

    function renderizarFavoritos() {
      const container = document.getElementById('favoritesListings');
      if (!container) return;

      const favoritos = state.anuncios.filter(a => state.favoritos.includes(a.id));

      container.innerHTML = '';

      if (!favoritos.length) {
        container.innerHTML = `
          <div class="empty-listings" style="grid-column:1/-1;">
            <i class="fa-solid fa-heart" style="font-size:25px;display:block;margin-bottom:8px;"></i>
            Nenhum favorito ainda.
          </div>
        `;
        return;
      }

      favoritos.forEach((anuncio) => {
        const fotoPrincipal = Array.isArray(anuncio.fotos) && anuncio.fotos.length
          ? anuncio.fotos[0]
          : 'https://images.unsplash.com/photo-1581578731548-c64695cc6952?auto=format&fit=crop&w=900&q=80';

        const card = document.createElement('article');
        card.className = 'listing-card';

        card.innerHTML = `
          <div class="listing-photo-container">
            <img class="listing-photo" src="${fotoPrincipal}" alt="${anuncio.titulo || 'Favorito'}">
          </div>

          <div class="listing-body">
            <div class="listing-price">${formatarPreco(anuncio.preco)}</div>
            <div class="listing-title">${anuncio.titulo || 'Anúncio sem título'}</div>
            <div class="listing-meta">${anuncio.cidade || 'Local'} • ${anuncio.estado || 'BR'}</div>
          </div>
        `;

        container.appendChild(card);
      });
    }

    function mostrarPreviaFotos(event) {
      const files = Array.from(event.target.files || []).slice(0, 5);
      const preview = document.getElementById('photoPreview');
      if (!preview) return;

      preview.innerHTML = '';
      state.fotosSelecionadas = [];

      files.forEach(file => {
        const reader = new FileReader();
        reader.onload = () => {
          state.fotosSelecionadas.push({ data: reader.result });

          const thumb = document.createElement('img');
          thumb.src = reader.result;
          preview.appendChild(thumb);
        };
        reader.readAsDataURL(file);
      });
    }

    function publicarServico(event) {
      event.preventDefault();

      const servTitle = document.getElementById('servTitle')?.value.trim();
      const servDesc = document.getElementById('servDesc')?.value.trim();
      const servPrice = document.getElementById('servPrice')?.value;

      if (!servTitle || !servDesc || !servPrice) {
        alert('Preencha título, descrição e preço antes de publicar.');
        return;
      }

      const anuncio = {
        id: gerarId(),
        tipo: document.getElementById('servType')?.value || 'Serviço',
        titulo: servTitle,
        categoria: document.getElementById('servCategory')?.value || 'Outros',
        preco: Number(servPrice),
        descricao: servDesc,
        contato: document.getElementById('servContact')?.value.trim() || '',
        cidade: state.localizacao.cidade !== 'TODAS' ? state.localizacao.cidade : 'Rio Branco',
        estado: state.localizacao.estado !== 'TODOS' ? state.localizacao.estado : 'AC',
        bairroArea: state.localizacao.bairroArea || '',
        fotos: state.fotosSelecionadas.map(item => item.data),
        autor: state.usuarioAtual?.email || 'visitante',
        criadoEm: new Date().toISOString()
      };

      state.anuncios.unshift(anuncio);
      localStorage.setItem(STORAGE_KEYS.listings, JSON.stringify(state.anuncios));

      document.getElementById('announceForm')?.reset();
      state.fotosSelecionadas = [];
      document.getElementById('photoPreview').innerHTML = '';

      renderizarAnuncios();
      renderizarFavoritos();
      document.querySelector('[data-target="homeView"]')?.click();
    }

    function abrirModalAuth() {
      document.getElementById('authModalOverlay')?.classList.add('active');
    }

    function fecharModalAuth() {
      document.getElementById('authModalOverlay')?.classList.remove('active');
    }

    function executarLogin(e) {
      e.preventDefault();

      const email = document.getElementById('loginEmail')?.value.trim();
      if (!email) return;

      state.usuarioAtual = {
        nome: email.split('@')[0],
        email
      };

      localStorage.setItem(STORAGE_KEYS.user, JSON.stringify(state.usuarioAtual));

      const nomeEl = document.getElementById('profileUserName');
      const emailEl = document.getElementById('profileUserEmail');

      if (nomeEl) nomeEl.textContent = state.usuarioAtual.nome || 'Usuário';
      if (emailEl) emailEl.textContent = state.usuarioAtual.email || 'Sem e-mail';

      fecharModalAuth();
      location.reload();
    }

    function atualizarPerfil() {
      const nomeEl = document.getElementById('profileUserName');
      const emailEl = document.getElementById('profileUserEmail');

      if (!state.usuarioAtual) {
        if (nomeEl) nomeEl.textContent = 'Visitante';
        if (emailEl) emailEl.textContent = 'Você não está logado';
        return;
      }

      if (nomeEl) nomeEl.textContent = state.usuarioAtual.nome || 'Usuário';
      if (emailEl) emailEl.textContent = state.usuarioAtual.email || 'Sem e-mail';
    }

    function inicializarFiltroCategoria() {
      document.querySelectorAll('.filter-pill').forEach((pill) => {
        pill.addEventListener('click', () => {
          state.categoriaAtual = pill.dataset.category || 'Todas';

          document.querySelectorAll('.filter-pill').forEach(btn => btn.classList.remove('active'));
          pill.classList.add('active');

          renderizarAnuncios();
        });
      });
    }

    function inicializarEventos() {
      document.querySelectorAll('.bottom-nav .nav-link').forEach(link => {
        link.addEventListener('click', (event) => {
          event.preventDefault();

          const target = link.getAttribute('data-target');
          document.querySelectorAll('.view-section').forEach(v => v.classList.remove('active'));
          document.querySelectorAll('.bottom-nav .nav-link').forEach(l => l.classList.remove('active'));

          document.getElementById(target)?.classList.add('active');
          link.classList.add('active');
        });
      });

      document.getElementById('announceForm')?.addEventListener('submit', publicarServico);
      document.getElementById('mainSearchInput')?.addEventListener('input', renderizarAnuncios);

      document.getElementById('themeToggle')?.addEventListener('click', () => {
        const nextTheme = document.body.getAttribute('data-theme') === 'dark' ? 'light' : 'dark';
        aplicarTema(nextTheme);
      });
    }

    function init() {
      state.usuarioAtual = safeParseJSON(STORAGE_KEYS.user, null);
      state.favoritos = safeParseJSON(STORAGE_KEYS.favorites, []);
      state.localizacao = safeParseJSON(STORAGE_KEYS.location, {
        estado: 'TODOS',
        cidade: 'TODAS',
        bairroArea: '',
        raioKm: 0
      });

      carregarDadosIniciais();
      inicializarFiltroLocalizacao();
      atualizarHeaderLocalizacaoText();
      atualizarPerfil();
      inicializarEventos();
      inicializarFiltroCategoria();

      aplicarTema(state.theme || 'light');
      renderizarAnuncios();
      renderizarFavoritos();
    }

    document.addEventListener('DOMContentLoaded', init);
  </script>
</body>
